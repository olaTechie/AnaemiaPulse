# --- Keywords Map ---
st.header("Keywords Map")
if 'keywords' in df.columns:
    def build_keyword_network(keywords_series):
        G = nx.Graph()
        for keywords in keywords_series.dropna():
            # Assuming keywords are separated by semicolons
            keyword_list = [kw.strip() for kw in keywords.split(';')]
            for i, kw in enumerate(keyword_list):
                for kw2 in keyword_list[i+1:]:
                    if G.has_edge(kw, kw2):
                        G[kw][kw2]['weight'] += 1
                    else:
                        G.add_edge(kw, kw2, weight=1)
        return G

    G = build_keyword_network(df['keywords'])
    
    # Plot using matplotlib (you could also use Plotly for interactive plots)
    plt.figure(figsize=(10, 8))
    pos = nx.spring_layout(G, k=0.5)
    # Normalize edge widths for better visualization
    edges = G.edges(data=True)
    weights = [data['weight'] for _, _, data in edges]
    nx.draw_networkx(G, pos, with_labels=True, width=[w*0.1 for w in weights], node_color='lightblue', font_size=8)
    st.pyplot(plt)
